150static void ReadUltraTask(void *pvParameter) {
151 while (true) {
152 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
153 if (MEDIR) {
154 DISTANCIA = HcSr04ReadDistanceInCentimeters();
155 }
156 }
157}

168static void LedsAndLcdTask(void *pvParameter) {
169 LcdItsE0803Off();
170 while (true) {
171 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
172 if (MEDIR) {
173 if (DISTANCIA < 10) {
174 LedsOffAll();
175 } else if (DISTANCIA >= 10 && DISTANCIA < 20) {
176 LedOn(LED_1);
177 LedOff(LED_2);
178 LedOff(LED_3);
179 } else if (DISTANCIA >= 20 && DISTANCIA <= 30) {
180 LedOn(LED_1);
181 LedOn(LED_2);
182 LedOff(LED_3);
183 } else if (DISTANCIA > 30) {
184 LedOn(LED_1);
185 LedOn(LED_2);
186 LedOn(LED_3);
187 }
188
189 if (!HOLD) {
190 LcdItsE0803Write(DISTANCIA);
191 }
192 } else {
193 LedsOffAll();
194 if (!HOLD) {
195 LcdItsE0803Off();
196 }
197 }
198 }
199}
