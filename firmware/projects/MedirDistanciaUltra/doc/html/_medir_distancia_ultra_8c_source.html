119static void LedsAndLcdTask(void *pvParameter) {
120 LcdItsE0803Off();
121 while (true) {
122 if (MEDIR) {
123 if (DISTANCIA < 10) {
124 LedsOffAll();
125 } else if (DISTANCIA >= 10 && DISTANCIA < 20) {
126 LedOn(LED_1);
127 LedOff(LED_2);
128 LedOff(LED_3);
129 } else if (DISTANCIA >= 20 && DISTANCIA <= 30) {
130 LedOn(LED_1);
131 LedOn(LED_2);
132 LedOff(LED_3);
133 } else if (DISTANCIA > 30) {
134 LedOn(LED_1);
135 LedOn(LED_2);
136 LedOn(LED_3);
137 }
138 if (!HOLD) {
139 LcdItsE0803Write(DISTANCIA);
140 }
141 } else {
142 LedsOffAll();
143 if (!HOLD) {
144 LcdItsE0803Off();
145 }
146 }
147 vTaskDelay(100 / portTICK_PERIOD_MS);
148 }
