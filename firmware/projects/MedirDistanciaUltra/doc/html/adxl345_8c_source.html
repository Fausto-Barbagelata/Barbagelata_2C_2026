67adxl345_status_t adxl345_write_register(uint8_t reg, uint8_t value)
68{
69 /* Una sola transaccion SPI: [addr|W][data]. CS bajo durante ambos bytes. */
70 uint8_t tx_buf[2];
71 tx_buf[0] = (reg & 0x3FU) | ADXL345_WRITE_BIT;
72 tx_buf[1] = value;
73
74 /* SpiWrite no necesita rx_buffer: los datos que el ADXL345 devuelve
75 * durante una escritura no son significativos y se descartan. */
76 SpiWrite(ADXL345_SPI_DEVICE, tx_buf, sizeof(tx_buf));
77
78 return ADXL345_OK;
79}

81adxl345_status_t adxl345_read_registers(uint8_t reg, uint8_t *rx_buf, uint8_t length)
82{
83 if ((rx_buf == NULL) || (length == 0U)) {
84 return ADXL345_ERROR_SPI;
85 }
86
87 /* Buffer de tx/rx de tamano length+1: el primer byte transmite la
88 * direccion+flags; los siguientes `length` bytes son "dummy" en tx
89 * (0x00) para generar los pulsos de reloj necesarios, y en rx
90 * contendran los datos reales devueltos por el sensor (el dato
91 * correspondiente al byte de direccion es descartado, ver Figura 38
92 * del datasheet: el primer byte de MISO durante la fase de
93 * direccionamiento no es valido). */
94 uint8_t tx_buf[9]; /* 1 (addr) + hasta 6 datos (X0,X1,Y0,Y1,Z0,Z1) + margen */
95 uint8_t rx_tmp[9];
96
97 if ((length + 1U) > sizeof(tx_buf)) {
98 return ADXL345_ERROR_SPI; /* longitud no soportada por el buffer estatico */
99 }
100
101 memset(tx_buf, 0x00, length + 1U);
102 tx_buf[0] = (reg & 0x3FU) | ADXL345_READ_BIT |
103 ((length > 1U) ? ADXL345_MULTIBYTE_BIT : 0x00U);
104
105 SpiReadWrite(ADXL345_SPI_DEVICE, tx_buf, rx_tmp, length + 1U);
106
107 /* rx_tmp[0] corresponde al byte de direccion (invalido); los datos
108 * utiles arrancan en rx_tmp[1]. */
109 memcpy(rx_buf, &rx_tmp[1], length);
110
111 return ADXL345_OK;
112}

114adxl345_status_t adxl345_init(const adxl345_config_t *config)
115{
116 if (config == NULL) {
117 return ADXL345_ERROR_SPI;
118 }
119
120 if (adxl345_configure_bus() != ADXL345_OK) {
121 return ADXL345_ERROR_SPI;
122 }
123
124 /* --- Autoverificacion: leer DEVID y comparar con 0xE5 ---
125 * Esto es critico en un dispositivo medico: detecta temprano un
126 * sensor mal soldado, un cable desconectado, o un bus SPI mal
127 * configurado, antes de iniciar una sesion de registro de varios dias. */
128 uint8_t devid = 0;
129 if (adxl345_read_registers(ADXL345_REG_DEVID, &devid, 1) != ADXL345_OK) {
130 return ADXL345_ERROR_SPI;
131 }
132 if (devid != ADXL345_DEVID_VALUE) {
133 return ADXL345_ERROR_DEVID;
134 }
135
136 /* --- Poner el sensor en standby antes de configurar ---
137 * Recomendado por el datasheet: configurar en standby y luego pasar
138 * a modo medicion, para evitar transitorios/ruido en las primeras
139 * muestras. */
140 adxl345_write_register(ADXL345_REG_POWER_CTL, 0x00U);
141
142 /* --- DATA_FORMAT: rango, resolucion, SPI 4 hilos --- */
143 uint8_t data_format = (config->range & ADXL345_RANGE_MASK);
144 if (config->full_resolution) {
145 data_format |= ADXL345_DATA_FORMAT_FULL_RES;
146 }
147 /* SPI_3WIRE bit se deja en 0 -> modo 4 hilos (SDI/SDO separados) */
148 adxl345_write_register(ADXL345_REG_DATA_FORMAT, data_format);
149
150 /* --- BW_RATE: ODR y modo de bajo consumo ---
151 * Para monitoreo ambulatorio de movimiento de laringe/cuello, una
152 * ODR de 25-100 Hz suele ser mas que suficiente y reduce
153 * drasticamente el consumo (ver Tabla 7/8 del datasheet: 100 Hz
154 * normal = 140 uA vs 12.5 Hz = 50 uA vs 12.5 Hz low-power = 34 uA). */
155 uint8_t bw_rate = (config->rate_code & ADXL345_BW_RATE_MASK);
156 if (config->low_power_mode) {
157 bw_rate |= ADXL345_BW_RATE_LOW_POWER;
158 }
159 adxl345_write_register(ADXL345_REG_BW_RATE, bw_rate);
160
161 /* --- FIFO: modo stream para minimizar transacciones SPI ---
162 * Acumular varias muestras en el FIFO del ADXL345 y leerlas en un
163 * solo burst reduce drasticamente la cantidad de veces que el MCU
164 * debe salir de modo de bajo consumo para atender al sensor,
165 * comparado con leer muestra a muestra a la ODR configurada. */
166 if (config->use_fifo_stream) {
167 uint8_t fifo_ctl = ADXL345_FIFO_MODE_STREAM |
168 (config->fifo_watermark_samples & ADXL345_FIFO_SAMPLES_MASK);
169 adxl345_write_register(ADXL345_REG_FIFO_CTL, fifo_ctl);
170
171 /* Habilitar interrupcion de watermark en INT1 para despertar al
172 * MCU solo cuando haya suficientes muestras acumuladas. */
173 adxl345_write_register(ADXL345_REG_INT_MAP, 0x00U); /* todo a INT1 */
174 adxl345_write_register(ADXL345_REG_INT_ENABLE, ADXL345_INT_WATERMARK);
175 } else {
176 adxl345_write_register(ADXL345_REG_FIFO_CTL, ADXL345_FIFO_MODE_BYPASS);
177 }
178
179 /* --- Entrar en modo medicion --- */
180 adxl345_write_register(ADXL345_REG_POWER_CTL, ADXL345_POWER_CTL_MEASURE);
181
182 s_active_config = *config;
183 s_initialized = true;
184
185 return ADXL345_OK;
186}

188adxl345_status_t adxl345_read_accel(adxl345_accel_raw_t *out)
189{
190 if (out == NULL) {
191 return ADXL345_ERROR_SPI;
192 }
193
194 /* Burst read de 6 bytes desde DATAX0: garantiza que X, Y, Z
195 * pertenecen a la misma muestra (evita "tearing" entre ejes que
196 * podria ocurrir si se leyeran registros individualmente mientras
197 * el sensor sigue actualizando datos). */
198 uint8_t raw[6];
199 adxl345_status_t status = adxl345_read_registers(ADXL345_REG_DATAX0, raw, 6);
200 if (status != ADXL345_OK) {
201 return status;
202 }
203
204 /* Formato: LSB primero (DATAx0), luego MSB (DATAx1), complemento a 2. */
205 out->x = (int16_t)((raw[1] << 8) | raw[0]);
206 out->y = (int16_t)((raw[3] << 8) | raw[2]);
207 out->z = (int16_t)((raw[5] << 8) | raw[4]);
208
209 return ADXL345_OK;
210}

212int32_t adxl345_raw_to_mg(int16_t raw, bool full_resolution, uint8_t range)
213{
214 /* En modo full-resolution la escala es siempre ~3.9 mg/LSB,
215 * independientemente del rango (Tabla 1 del datasheet). */
216 if (full_resolution) {
217 return ((int32_t)raw * 39) / 10; /* 3.9 mg/LSB */
218 }
219
220 /* En modo 10-bit fijo, la escala depende del rango seleccionado. */
221 switch (range & ADXL345_RANGE_MASK) {
222 case ADXL345_RANGE_2G: return ((int32_t)raw * 39) / 10; /* 3.9 mg/LSB */
223 case ADXL345_RANGE_4G: return ((int32_t)raw * 78) / 10; /* 7.8 mg/LSB */
224 case ADXL345_RANGE_8G: return ((int32_t)raw * 156) / 10; /* 15.6 mg/LSB */
225 case ADXL345_RANGE_16G: return ((int32_t)raw * 312) / 10; /* 31.2 mg/LSB */
226 default: return 0;
227 }
228}

230adxl345_status_t adxl345_get_fifo_entries(uint8_t *entries)
231{
232 if (entries == NULL) {
233 return ADXL345_ERROR_SPI;
234 }
235
236 uint8_t fifo_status = 0;
237 adxl345_status_t status = adxl345_read_registers(ADXL345_REG_FIFO_STATUS, &fifo_status, 1);
238 if (status != ADXL345_OK) {
239 return status;
240 }
241
242 *entries = fifo_status & 0x3FU; /* bits D5:D0 = Entries */
243 return ADXL345_OK;
244}

246adxl345_status_t adxl345_standby(void)
247{
248 /* Limpiar el bit MEASURE preserva el contenido del FIFO y baja el
249 * consumo a ~0.1 uA tipico: ideal para pausas largas en el
250 * monitoreo sin perder la configuracion del sensor. */
251 return adxl345_write_register(ADXL345_REG_POWER_CTL, 0x00U);
252}
